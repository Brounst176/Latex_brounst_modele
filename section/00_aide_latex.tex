    \descrVarEq{S=[X_S,Y_S,Z_S]}{Coordonnée du point de la caméra dans $\scu^{glob}$}\\
    R_{glob}^{cam}=R_\omega\cdot R_\phi\cdot R_\kappa=\begin{bmatrix}
    r_{11}&r_{21}&r_{31}\\
